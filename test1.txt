class Solution {

    static final long MOD = 1000000007;

    public int countOfArrays(int n, int l, int r) {

        int m = r - l + 1;
        int size = 2 * m;

        // Transition matrix
        long[][] matrix = new long[size][size];

        // up[i] depends on previous down[j], where j < i
        for (int i = 0; i < m; i++) {

            for (int j = 0; j < i; j++) {
                matrix[i][m + j] = 1;
            }
        }

        // down[i] depends on previous up[j], where j > i
        for (int i = 0; i < m; i++) {

            for (int j = i + 1; j < m; j++) {
                matrix[m + i][j] = 1;
            }
        }

        // Initial DP vector for length 2
        long[] dp = new long[size];

        for (int i = 0; i < m; i++) {
            dp[i] = i;
            dp[m + i] = m - 1 - i;
        }

        // Apply matrix exponentiation n - 2 times
        int power = n - 2;

        while (power > 0) {

            if ((power & 1) == 1) {
                dp = multiply(matrix, dp);
            }

            matrix = multiply(matrix, matrix);

            power >>= 1;
        }

        // Calculate final answer
        long ans = 0;

        for (int i = 0; i < size; i++) {
            ans = (ans + dp[i]) % MOD;
        }

        return (int) ans;
    }

    // Matrix multiplication
    private long[][] multiply(long[][] a, long[][] b) {

        int size = a.length;

        long[][] result = new long[size][size];

        for (int i = 0; i < size; i++) {

            for (int k = 0; k < size; k++) {

                if (a[i][k] == 0) {
                    continue;
                }

                for (int j = 0; j < size; j++) {

                    result[i][j] = (
                        result[i][j] +
                        a[i][k] * b[k][j]
                    ) % MOD;
                }
            }
        }

        return result;
    }

    // Matrix and vector multiplication
    private long[] multiply(long[][] a, long[] b) {

        int size = b.length;

        long[] result = new long[size];

        for (int i = 0; i < size; i++) {

            for (int j = 0; j < size; j++) {

                result[i] = (
                    result[i] + a[i][j] * b[j]
                ) % MOD;
            }
        }

        return result;
    }
}
